```python
# 连接 vSphere 6.5/7.0/8.0
from pyVim.connect import SmartConnect, Disconnect
from pyVmomi import vim

class VSphereAdapter(PlatformAdapter):
    async def connect(self, config: PlatformConfig) -> bool:
        context = ssl._create_unverified_context()
        self._si = SmartConnect(
            host=config.host,
            user=config.username,
            pwd=config.password,
            port=config.port,
            sslContext=context,
        )
        self._content = self._si.RetrieveContent()
        return True
    
    async def list_vms(self) -> list[DeviceInfo]:
        container = self._content.viewManager.CreateContainerView(
            self._content.rootFolder, [vim.VirtualMachine], True
        )
        vms = []
        for vm in container.view:
            vms.append(DeviceInfo(
                id=vm._moId,
                name=vm.name,
                status="running" if vm.runtime.powerState == vim.VirtualMachinePowerState.poweredOn else "stopped",
                ip_address=vm.guest.ipAddress if vm.guest else "",
                platform="vSphere",
            ))
        container.Destroy()
        return vms
```

```python
import libvirt

class KVMAdapter(PlatformAdapter):
    async def connect(self, config: PlatformConfig) -> bool:
        uri = f"qemu+ssh://{config.username}@{config.host}/system"
        self._conn = libvirt.open(uri)
        return self._conn is not None
    
    async def list_vms(self) -> list[DeviceInfo]:
        domains = self._conn.listAllDomains()
        vms = []
        for dom in domains:
            vms.append(DeviceInfo(
                id=dom.UUIDString(),
                name=dom.name(),
                status="running" if dom.isActive() else "stopped",
                platform="KVM",
            ))
        return vms
```

```python
import httpx

class FusionSphereAdapter(PlatformAdapter):
    async def connect(self, config: PlatformConfig) -> bool:
        self._client = httpx.AsyncClient(
            base_url=f"https://{config.host}:{config.port}/service",
            verify=config.verify_ssl,
            timeout=30.0,
        )
        response = await self._client.post("/sessions", json={
            "username": config.username,
            "password": config.password,
        })
        response.raise_for_status()
        self._token = response.json().get("token")
        self._client.headers["X-Auth-Token"] = self._token
        return True
    
    async def list_vms(self) -> list[DeviceInfo]:
        response = await self._client.get("/vms")
        response.raise_for_status()
        vms_data = response.json().get("vms", [])
        return [DeviceInfo(
            id=vm.get("vmId"),
            name=vm.get("name"),
            status=vm.get("status"),
            platform="FusionSphere",
        ) for vm in vms_data]
```

**迁移场景**:

| 源 | 目标 | 工具 | 步骤 |
|----|------|------|------|
| VMware | KVM | virt-v2v | 导出 → 转换 → 导入 |
| VMware | FusionSphere | qemu-img | 导出 → 转换 → 上传 |
| KVM | VMware | qemu-img + ovftool | 导出 → 转换 → 导入 |

**迁移引擎设计**:
```python
class MigrationEngine:
    async def plan_migration(self, vm_id: str, source: str, target: str) -> MigrationPlan:
        # 1. 获取源 VM 信息
        source_vm = await self._get_vm_info(vm_id, source)
        
        # 2. 分析兼容性
        compatibility = self._analyze_compatibility(source_vm, source, target)
        
        # 3. 选择迁移工具
        tool = self._select_tool(source, target, source_vm)
        
        # 4. 生成迁移计划
        return MigrationPlan(
            source_vm=source_vm,
            target_platform=target,
            tool=tool,
            steps=self._generate_steps(source_vm, tool),
            estimated_time=self._estimate_time(source_vm, tool),
            risks=compatibility.risks,
        )
    
    def _select_tool(self, source: str, target: str, vm_info: dict) -> str:
        if source == "vsphere" and target == "kvm":
            return "virt-v2v"  # Red Hat 官方工具
        elif source == "vsphere" and target == "fusionsphere":
            return "qemu-img"  # 磁盘格式转换
        elif source == "kvm" and target == "vsphere":
            return "qemu-img"  # 转换后导入
```

## 4. 数据库变更

### 4.1 新增表

```sql
-- 平台配置表
CREATE TABLE platforms (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    platform_type VARCHAR(50) NOT NULL,  -- vsphere, kvm, fusionsphere
    host VARCHAR(255) NOT NULL,

    status VARCHAR(50) DEFAULT 'disconnected',
    last_sync_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW(),

CREATE TABLE migration_tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    source_platform VARCHAR(50) NOT NULL,
    target_platform VARCHAR(50) NOT NULL,
    target_host VARCHAR(255),
    migration_tool VARCHAR(50) NOT NULL,
    status VARCHAR(50) DEFAULT 'pending',
    plan JSONB NOT NULL,
    progress INTEGER DEFAULT 0,
    error_message TEXT,
    started_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW()
);
```
